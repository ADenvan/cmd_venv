5. КАК ИСПОЛЬЗОВАТЬ КАК ШАБЛОН ДЛЯ БАЗЫ ДАННЫХ
# Предлагаемая структура Django моделей на основе файла:
class Employee:
    - id
    - full_name (ФИО)
    - position (Должность)
    - department (Отдел)
    - hire_date (Дата приема)
    - payment_type (тип оплаты: 'salary' | 'hourly')
    - base_salary (оклад/часовая ставка)
class WorkPeriod:
    - employee (FK)
    - period_start (начало периода)
    - period_end (конец периода)
    - working_days (рабочих дней)
    - actual_hours (отработано часов)
    - hourly_rate (часовая ставка)
class PayrollCalculation:
    - employee (FK)
    - period (FK)
    - base_pay (основное начисление)
    - bonuses (премии)
    - deductions (удержания)
    - total_accrued (всего начислено)
    - tax_13 (НДФЛ 13%)
    - total_payable (к выдаче)
